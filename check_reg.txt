    // instructions encode the register number as raw byte(s) that come
    // directly after the opcode — NOT as digit keypresses that go through
    // the two-digit pending-reg accumulator:
    //
    //   registers  0–9  → 1 byte  whose value IS the register (0–9, ≤ 9)

    //                      (e.g. 12 = register 12, even though 12 = KC_B)
    //   registers 20–99 → 2 bytes both ≤ 9 (e.g. [8,1] = register 81)
